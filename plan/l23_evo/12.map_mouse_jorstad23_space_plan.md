# Script 12: Map mouse L2/3 IT (Yoo25) into human VX archetype space

## Context
Scripts 09–11 define a human L2/3 IT archetype space via varimax rotation (VX2/VX6–VX10) and PCHA (NOC=4). Script 12 maps mouse L2/3 IT cells (Yoo25, P21) into this space by: (1) identifying high-loading genes per VX component, (2) finding their mouse orthologs, (3) computing per-mouse-cell similarity to all human cells using those shared genes, and (4) embedding each mouse cell as a weighted average of its top-k most similar human cells' coordinates.

`data/human_mouse_orthologs.tsv` — BioMart ortholog table downloaded via the BioMart REST API using `requests` (no extra packages needed). Script 12 downloads and caches it automatically on first run:
```python
import requests, io
if not os.path.exists(IN_ORTHOLOGS):
    xml = """<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE Query>
<Query virtualSchemaName="default" formatter="TSV" header="1" uniqueRows="1" datasetConfigVersion="0.6">
  <Dataset name="hsapiens_gene_ensembl" interface="default">
    <Attribute name="external_gene_name"/>
    <Attribute name="mmusculus_homolog_associated_gene_name"/>
  </Dataset>
</Query>"""
    # NOTE: must use external_gene_name (not hgnc_symbol) — BioMart forbids attributes
    # from multiple attribute pages in one query, and hgnc_symbol is on a different page.
    r = requests.get('https://www.ensembl.org/biomart/martservice', params={'query': xml})
    r.raise_for_status()
    df = pd.read_csv(io.StringIO(r.text), sep='\t')
    df.columns = ['human_symbol', 'mouse_symbol']
    df = df.dropna().query("human_symbol != '' and mouse_symbol != ''").drop_duplicates()
    df.to_csv(IN_ORTHOLOGS, sep='\t', index=False)
    print(f'Saved {IN_ORTHOLOGS}  ({len(df)} orthologs)')
```

## Implementation Steps

### 1. Select high-loading genes from VX subspace
```python
load_df = pd.read_csv(IN_LOADINGS, sep='\t', index_col=0)   # (2000, 10)
seen, vx_genes = set(), []
for col in VX_COLS:
    for gene in load_df[col].abs().nlargest(N_TOP_GENES_PER_VX).index:
        if gene not in seen:
            seen.add(gene)
            vx_genes.append(gene)
# vx_genes: up to 300 human gene symbols
```

### 2. Load ortholog table and map to mouse symbols
```python
ortho = pd.read_csv(IN_ORTHOLOGS, sep='\t')
# keep 1-to-1 mappings only (drop duplicates in both directions)
ortho = ortho.drop_duplicates('human_symbol').drop_duplicates('mouse_symbol')
h2m = ortho.set_index('human_symbol')['mouse_symbol'].to_dict()
# pre-filter vx_genes to those with a mouse ortholog
gene_pairs_pre = [(g, h2m[g]) for g in vx_genes if g in h2m]
# further filter to genes actually present in both h5ad var_names
# (some orthologs are absent from the dataset gene lists)
gene_pairs  = [(hg, mg) for hg, mg in gene_pairs_pre if hg in h_gene_set and mg in m_gene_set]
human_genes = [p[0] for p in gene_pairs]
mouse_genes = [p[1] for p in gene_pairs]
# In practice: 214 VX genes → 193 with orthologs → 188 present in both datasets
```

### 3. Load human and mouse adata (needed to build gene-name sets before step 2 filtering)
```python
h_adata      = ad.read_h5ad(INPUT_HUMAN)
h_gene_names = h_adata.var['feature_name'].values if 'feature_name' in h_adata.var.columns else h_adata.var_names.values
h_gene_set   = set(h_gene_names)
types        = h_adata.obs[CLUSTER_COL].values

m_adata      = ad.read_h5ad(INPUT_MOUSE)
m_adata      = m_adata[m_adata.obs['Subclass'] == MOUSE_SUBCLASS]   # 2213 cells
m_gene_names = m_adata.var_names.values
m_gene_set   = set(m_gene_names)
```
Both adatas must be loaded before building `gene_pairs` (step 2 above) so that `h_gene_set` and `m_gene_set` are available for the within-dataset filter.

After `gene_pairs` is finalized, extract the expression submatrices:
```python
h_idx      = np.array([np.where(h_gene_names == g)[0][0] for g in human_genes])
X_human    = h_adata.X[:, h_idx].toarray().astype(np.float32)
# h_adata.X is already log-normalized (confirmed in script 01) — use as-is
```

### 4. Normalize mouse expression (selected ortholog genes only)
```python
m_idx        = np.array([np.where(m_gene_names == g)[0][0] for g in mouse_genes])
X_mouse_raw  = m_adata.X[:, m_idx].toarray().astype(np.float32)
# normalize raw counts → log2(CP10k + 1)
depths       = X_mouse_raw.sum(axis=1, keepdims=True)
depths[depths == 0] = 1
X_mouse      = np.log2(X_mouse_raw / depths * 1e4 + 1)
```

### 5. Compute cosine similarity — chunked to avoid full matrix in memory
Cosine similarity is fast (pure BLAS matrix multiply, no ranking) and avoids the ~417 MB full pairwise matrix by processing mouse cells in chunks and retaining only top-k indices and scores per chunk.

```python
# L2-normalize each cell's expression vector
def l2norm(X):
    norms = np.linalg.norm(X, axis=1, keepdims=True)
    norms[norms == 0] = 1
    return X / norms

X_human_n = l2norm(X_human)   # (n_human, n_genes)
X_mouse_n = l2norm(X_mouse)   # (n_mouse, n_genes)

n_mouse = X_mouse_n.shape[0]
topk_indices = np.empty((n_mouse, K_NEIGHBORS), dtype=np.int32)
topk_scores  = np.empty((n_mouse, K_NEIGHBORS), dtype=np.float32)

for start in range(0, n_mouse, CHUNK_SIZE):
    chunk = X_mouse_n[start:start + CHUNK_SIZE]        # (chunk, n_genes)
    sim_chunk = chunk @ X_human_n.T                    # (chunk, n_human)
    idx = np.argpartition(sim_chunk, -K_NEIGHBORS, axis=1)[:, -K_NEIGHBORS:]
    scores = np.take_along_axis(sim_chunk, idx, axis=1)
    topk_indices[start:start + len(chunk)] = idx
    topk_scores[start:start + len(chunk)]  = scores
```
Memory per chunk: `CHUNK_SIZE × n_human × 4 bytes` = 500 × 47125 × 4 ≈ 94 MB — fine.

### 6. Load saved PCHA results from script 09
```python
xp_human = pd.read_csv(IN_XP, sep='\t', index_col=0).values   # (n_human, NDIM)
aa        = pd.read_csv(IN_AA, sep='\t', index_col=0).values.T # (NDIM, NOC)
```

### 7. Embed mouse cells via weighted average of top-k human neighbors
```python
weights  = topk_scores / topk_scores.sum(axis=1, keepdims=True)   # (n_mouse, k)
mouse_xp = np.stack([
    (weights[i] @ xp_human[topk_indices[i]])
    for i in range(n_mouse)
])   # (n_mouse, NDIM)

mouse_coords_df = pd.DataFrame(mouse_xp, index=m_adata.obs_names,
                                columns=[f'PC{i+1}' for i in range(xp_human.shape[1])])
mouse_coords_df['Subclass']    = m_adata.obs['Subclass'].values
mouse_coords_df['Type_leiden'] = m_adata.obs['Type_leiden'].values
mouse_coords_df.to_csv(OUT_MOUSE_COORDS, sep='\t')
```

### 8. Visualize: human (downsampled) + mouse overlaid in PC1/PC2/PC3
```python
rng  = np.random.default_rng(0)
hidx = rng.choice(len(xp_human), N_DOWNSAMPLE_VIZ, replace=False)
xp_combined   = np.vstack([xp_human[hidx], mouse_xp])
cell_metadata = {
    'species':    np.array(['human'] * N_DOWNSAMPLE_VIZ + ['mouse'] * n_mouse),
    'Type_leiden': np.concatenate([types[hidx], m_adata.obs['Type_leiden'].values]),
}
scatter_categorical_html(
    xp_grid=[xp_combined], cell_metadata=cell_metadata,
    title=f'Jorstad23 (human) + Yoo25 L2/3 (mouse) — reprojected VX subspace',
    out_path=OUT_HTML,

    panel_3d=(0,1,2,'PC1','PC2','PC3'),

```
